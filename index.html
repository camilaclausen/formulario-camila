  // ══════════════════════════════════════════════════

  const MAILERLITE_API_KEY = 'Content-Type: application/json

                    Accept: application/json';

  const MAILERLITE_GROUP_ID = '186416148445660928;

  // ══════════════════════════════════════════════════


  const totalSteps = 4;

  let current = 1;


  function buildDots() {

    const ind = document.getElementById('stepIndicator');

    ind.innerHTML = '';

    for (let i = 1; i <= totalSteps; i++) {

      const d = document.createElement('div');

      d.className = 'step-dot' + (i === current ? ' active' : (i < current ? ' done' : ''));

      ind.appendChild(d);

    }

  }


  function updateProgress(step) {

    const pct = step > totalSteps ? 100 : Math.round((step / totalSteps) * 100);

    document.getElementById('progressBar').style.width = pct + '%';

  }


  function showStep(n) {

    document.querySelectorAll('.step').forEach(s => s.classList.remove('active'));

    document.querySelector('[data-step="' + n + '"]').classList.add('active');

    current = n;

    buildDots();

    updateProgress(n);

    window.scrollTo(0, 0);

  }


  function selectRadio(el, name) {

    document.querySelectorAll('[name="' + name + '"]').forEach(r => {

      r.closest('.radio-option').classList.remove('selected');

    });

    el.classList.add('selected');

    el.querySelector('input[type="radio"]').checked = true;

    const fg = el.closest('[id^="fg-"]');

    if (fg) fg.classList.remove('has-error');

  }


  function val(id) { const e = document.getElementById(id); return e ? e.value.trim() : ''; }

  function radio(name) { const c = document.querySelector('[name="' + name + '"]:checked'); return c ? c.value : ''; }


  function validateStep(n) {

    let ok = true;

    function chkText(id, fgId) {

      const fg = document.getElementById(fgId);

      if (fg) { if (!val(id)) { fg.classList.add('has-error'); ok = false; } else fg.classList.remove('has-error'); }

    }

    function chkEmail(id, fgId) {

      const fg = document.getElementById(fgId);

      if (fg) { const valid = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(val(id)); if (!valid) { fg.classList.add('has-error'); ok = false; } else fg.classList.remove('has-error'); }

    }

    function chkRadio(name, fgId) {

      const fg = document.getElementById(fgId);

      if (fg) { if (!radio(name)) { fg.classList.add('has-error'); ok = false; } else fg.classList.remove('has-error'); }

    }

    if (n === 1) { chkText('nombre','fg-nombre'); chkEmail('email','fg-email'); chkText('whatsapp','fg-whatsapp'); }

    if (n === 2) { chkRadio('tipo_negocio','fg-tipo_negocio'); chkText('descripcion','fg-descripcion'); chkRadio('facturacion','fg-facturacion'); }

    if (n === 3) { chkText('vision','fg-vision'); chkText('aguja','fg-aguja'); chkRadio('compromiso','fg-compromiso'); }

    if (n === 4) { chkRadio('servicio','fg-servicio'); chkRadio('inversion','fg-inversion'); chkRadio('inicio','fg-inicio'); }

    return ok;

  }


  function goTo(n) {

    if (n > current && !validateStep(current)) return;

    showStep(n);

  }


  async function submitForm() {

    if (!validateStep(4)) return;

    const btn = document.getElementById('btnSubmit');

    const errBox = document.getElementById('apiError');

    btn.disabled = true;

    btn.textContent = 'Enviando...';

    errBox.style.display = 'none';


    const payload = {

      email: val('email'),

      fields: {

        name: val('nombre'),

        whatsapp: val('whatsapp'),

        pagina_web: val('web'),

        instagram: val('instagram'),

        linkedin: val('linkedin'),

        otra_red_social: val('otra_red'),

        tipo_negocio: radio('tipo_negocio'),

        descripcion_negocio: val('descripcion'),

        facturacion_anual: radio('facturacion'),

        vision_6_meses: val('vision'),

        mueve_la_aguja: val('aguja'),

        compromiso: radio('compromiso'),

        servicio_interes: radio('servicio'),

        inversion_dispuesta: radio('inversion'),

        cuando_empezar: radio('inicio')

      },

      groups: [186416148445660928],

      status: 'active'

    };


    try {

      const res = await fetch('https://connect.mailerlite.com/api/subscribers', {

        method: 'POST',

        headers: {

          'Content-Type': 'application/json',

          'Authorization': 'Bearer ' + Content-Type: application/json

                    Accept: application/json,

          'Accept': 'application/json'

        },

        body: JSON.stringify(payload)

      });


      if (res.ok || res.status === 201 || res.status === 200) {

        showStep(5);

      } else {

        const data = await res.json();

        console.error('MailerLite error:', data);

        errBox.style.display = 'block';

        btn.disabled = false;

        btn.textContent = 'Enviar aplicación';

      }

    } catch (err) {

      console.error('Network error:', err);

      errBox.style.display = 'block';

      btn.disabled = false;

      btn.textContent = 'Enviar aplicación';
